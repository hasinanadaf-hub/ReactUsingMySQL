step 5
Open server.js file and add following code
const  express= require("express")
const cors = require("cors");
const mysql = require("mysql2")
const bodyparser =require("body-parser")

const myserver = express();

myserver.listen(7860,()=>{
    console.log("Server started at port 7860")
});

const db = mysql.createConnection({
    host:"localhost",
    user:"root",
    password:"root@1234",
    database:"hasina",
    port:3306
});

db.connect((err)=>{
    if(err)
    {
    console.error('Error connecting to database:', err);
        return;
    }
    console.log('Connected to database');
});

myserver.get("/api/employee/showall",(req,res)=>{
    const sql = "SELECT * from employee";
    db.query(sql,(err,result)=>{
        if(err)
        {
            console.log("Not able to fetch data",err);
            res.send("Error Fetching Employee Data")
        }
        else
        {
            res.send({status:"Ok",data:result});
        }
    })
})
Run this url on brower (http://localhost:7860/api/employee/showall)

import React from 'react'
import {BrowserRouter,Route,Routes} from 'react-router-dom'
import ViewEmployee from './pages/ViewEmployee'
import CreateEmployee from './pages/CreateEmployee'
import EditEmployee from './pages/EditEmployee'
import DispplayEmployee from './pages/DispplayEmployee'
function App() {
  return (
    <>
      <BrowserRouter>
        <Routes>
          <Route path='/' element={<ViewEmployee/>} />
          <Route path="/createemp" element={<CreateEmployee/>}/>
          <Route path="/editemp/:eid" element={<EditEmployee/>}/>
          <Route path="/displayemp/:eid" element={<DispplayEmployee/>}/>
        </Routes>
      </BrowserRouter>
    </>
  )
}

export default App
